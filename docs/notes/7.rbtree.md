# 第 8 步：rbtree 引擎（红黑树）

> 第三个引擎。和 array、hash 接口完全一样，但底层是一棵自平衡二叉搜索树。
> key 按字典序排列，查找/插入/删除都是 O(log n)。

---

## 为什么需要红黑树？

| 引擎 | 查找复杂度 | 有序？ | 特点 |
|---|---|---|---|
| array | O(n) | 按插入顺序 | 最简单 |
| hash | O(1) 平均 | ❌ 无序 | 最快查找 |
| **rbtree** | **O(log n)** | **✅ 按 key 排序** | 有序遍历、范围查询 |

红黑树是一棵**自平衡二叉搜索树**：
- key 按字典序排列（左子树 < 根 < 右子树）
- 通过旋转和变色保持平衡，不会退化成链表
- 查找/插入/删除都是 O(log n)

---

## 头文件声明

```cpp
// ---------------- rbtree 引擎 ----------------
#define RB_RED    1
#define RB_BLACK  2

struct RbNode {
    unsigned char color;
    RbNode *left;
    RbNode *right;
    RbNode *parent;
    char *key;
    char *value;
};

struct RbTree {
    RbNode *root;
    RbNode *nil;      // NIL 哨兵节点
    int count;
};

extern RbTree g_tree;

int  kvstore_rbtree_create(RbTree *tree);
void kvstore_rbtree_destroy(RbTree *tree);
int  kvs_rbtree_set(RbTree *tree, const char *key, const char *value);
const char *kvs_rbtree_get(RbTree *tree, const char *key);
int  kvs_rbtree_delete(RbTree *tree, const char *key);
int  kvs_rbtree_modify(RbTree *tree, const char *key, const char *value);
int  kvs_rbtree_count(RbTree *tree);
```

## 完整实现 `kvstore_rbtree.cpp`

```cpp
#include "kvstore.hpp"

RbTree g_tree;

// ============ 旋转 ============

// 左旋：x 的右孩子 y 提到 x 的位置
static void rbtree_left_rotate(RbTree *T, RbNode *x) {
    RbNode *y = x->right;
    x->right = y->left;
    if (y->left != T->nil) y->left->parent = x;
    y->parent = x->parent;
    if (x->parent == T->nil)      T->root = y;
    else if (x == x->parent->left) x->parent->left = y;
    else                           x->parent->right = y;
    y->left = x;
    x->parent = y;
}

// 右旋：y 的左孩子 x 提到 y 的位置
static void rbtree_right_rotate(RbTree *T, RbNode *y) {
    RbNode *x = y->left;
    y->left = x->right;
    if (x->right != T->nil) x->right->parent = y;
    x->parent = y->parent;
    if (y->parent == T->nil)      T->root = x;
    else if (y == y->parent->right) y->parent->right = x;
    else                            y->parent->left = x;
    x->right = y;
    y->parent = x;
}

// ============ 插入 ============

static void rbtree_insert_fixup(RbTree *T, RbNode *z) {
    while (z->parent->color == RB_RED) {
        if (z->parent == z->parent->parent->left) {
            RbNode *y = z->parent->parent->right;
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->right) {
                    z = z->parent;
                    rbtree_left_rotate(T, z);
                }
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                rbtree_right_rotate(T, z->parent->parent);
            }
        } else {
            RbNode *y = z->parent->parent->left;
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->left) {
                    z = z->parent;
                    rbtree_right_rotate(T, z);
                }
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                rbtree_left_rotate(T, z->parent->parent);
            }
        }
    }
    T->root->color = RB_BLACK;
}

static void rbtree_insert(RbTree *T, RbNode *z) {
    RbNode *y = T->nil;
    RbNode *x = T->root;
    while (x != T->nil) {
        y = x;
        int cmp = strcmp(z->key, x->key);
        if (cmp < 0)      x = x->left;
        else if (cmp > 0) x = x->right;
        else return;   // key 已存在
    }
    z->parent = y;
    if (y == T->nil)              T->root = z;
    else if (strcmp(z->key, y->key) < 0) y->left = z;
    else                           y->right = z;
    z->left = T->nil;
    z->right = T->nil;
    z->color = RB_RED;
    rbtree_insert_fixup(T, z);
}

// ============ 搜索 ============

static RbNode *rbtree_search(RbTree *T, const char *key) {
    RbNode *node = T->root;
    while (node != T->nil) {
        int cmp = strcmp(key, node->key);
        if (cmp < 0)      node = node->left;
        else if (cmp > 0) node = node->right;
        else              return node;
    }
    return T->nil;
}

static RbNode *rbtree_minimum(RbTree *T, RbNode *x) {
    while (x->left != T->nil) x = x->left;
    return x;
}

// 找 x 的中序后继
static RbNode *rbtree_successor(RbTree *T, RbNode *x) {
    if (x->right != T->nil) {
        return rbtree_minimum(T, x->right);
    }
    RbNode *y = x->parent;
    while ((y != T->nil) && (x == y->right)) {
        x = y;
        y = x->parent;
    }
    return y;
}

// ============ 删除（完整版） ============

static void rbtree_delete_fixup(RbTree *T, RbNode *x) {
    while ((x != T->root) && (x->color == RB_BLACK)) {
        if (x == x->parent->left) {
            RbNode *w = x->parent->right;
            if (w->color == RB_RED) {
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                rbtree_left_rotate(T, x->parent);
                w = x->parent->right;
            }
            if ((w->left->color == RB_BLACK) && (w->right->color == RB_BLACK)) {
                w->color = RB_RED;
                x = x->parent;
            } else {
                if (w->right->color == RB_BLACK) {
                    w->left->color = RB_BLACK;
                    w->color = RB_RED;
                    rbtree_right_rotate(T, w);
                    w = x->parent->right;
                }
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->right->color = RB_BLACK;
                rbtree_left_rotate(T, x->parent);
                x = T->root;
            }
        } else {
            RbNode *w = x->parent->left;
            if (w->color == RB_RED) {
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                rbtree_right_rotate(T, x->parent);
                w = x->parent->left;
            }
            if ((w->left->color == RB_BLACK) && (w->right->color == RB_BLACK)) {
                w->color = RB_RED;
                x = x->parent;
            } else {
                if (w->left->color == RB_BLACK) {
                    w->right->color = RB_BLACK;
                    w->color = RB_RED;
                    rbtree_left_rotate(T, w);
                    w = x->parent->left;
                }
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->left->color = RB_BLACK;
                rbtree_right_rotate(T, x->parent);
                x = T->root;
            }
        }
    }
    x->color = RB_BLACK;
}

static RbNode *rbtree_delete(RbTree *T, RbNode *z) {
    RbNode *y, *x;

    if ((z->left == T->nil) || (z->right == T->nil)) {
        y = z;
    } else {
        y = rbtree_successor(T, z);
    }

    if (y->left != T->nil)      x = y->left;
    else if (y->right != T->nil) x = y->right;
    else                          x = T->nil;

    x->parent = y->parent;
    if (y->parent == T->nil)       T->root = x;
    else if (y == y->parent->left) y->parent->left = x;
    else                           y->parent->right = x;

    if (y != z) {
        char *tmp = z->key;
        z->key = y->key;
        y->key = tmp;
        tmp = z->value;
        z->value = y->value;
        y->value = tmp;
    }

    if (y->color == RB_BLACK) {
        rbtree_delete_fixup(T, x);
    }

    return y;
}

// ============ KV 引擎接口 ============

int kvstore_rbtree_create(RbTree *tree) {
    if (!tree) return -1;
    tree->nil = new RbNode;
    tree->nil->color = RB_BLACK;
    tree->nil->left = tree->nil->right = tree->nil->parent = nullptr;
    tree->nil->key = new char[1]();
    tree->root = tree->nil;
    tree->count = 0;
    return 0;
}

void kvstore_rbtree_destroy(RbTree *tree) {
    if (!tree) return;
    RbNode *node;
    while ((node = rbtree_minimum(tree, tree->root)) != tree->nil) {
        RbNode *dead = rbtree_delete(tree, node);
        if (dead) {
            delete[] dead->key;
            delete[] dead->value;
            delete dead;
        }
    }
    delete[] tree->nil->key;
    delete tree->nil;
    tree->root = tree->nil = nullptr;
    tree->count = 0;
}

int kvs_rbtree_set(RbTree *tree, const char *key, const char *value) {
    RbNode *exist = rbtree_search(tree, key);
    if (exist != tree->nil) {
        delete[] exist->value;
        exist->value = new char[strlen(value) + 1]();
        strcpy(exist->value, value);
        return 0;
    }
    RbNode *node = new RbNode;
    node->key = new char[strlen(key) + 1]();
    node->value = new char[strlen(value) + 1]();
    strcpy(node->key, key);
    strcpy(node->value, value);
    rbtree_insert(tree, node);
    tree->count++;
    return 0;
}

const char *kvs_rbtree_get(RbTree *tree, const char *key) {
    RbNode *node = rbtree_search(tree, key);
    if (node == tree->nil) return nullptr;
    return node->value;
}

int kvs_rbtree_delete(RbTree *tree, const char *key) {
    RbNode *node = rbtree_search(tree, key);
    if (node == tree->nil) return tree->count;
    RbNode *dead = rbtree_delete(tree, node);
    if (dead) {
        delete[] dead->key;
        delete[] dead->value;
        delete dead;
    }
    tree->count--;
    return 0;
}

int kvs_rbtree_modify(RbTree *tree, const char *key, const char *value) {
    RbNode *node = rbtree_search(tree, key);
    if (node == tree->nil) return tree->count;
    delete[] node->value;
    node->value = new char[strlen(value) + 1]();
    strcpy(node->value, value);
    return 0;
}

int kvs_rbtree_count(RbTree *tree) {
    if (!tree) return -1;
    return tree->count;
}
```

## 核心知识点

### NIL 哨兵节点

普通二叉树判断空指针用 `if (node == nullptr)`，每个指针都要判断。红黑树用一个**统一的 NIL 节点**代替所有空指针：

```
普通树：  root->left == nullptr？要判断
红黑树：  root->left == nil？  统一指向同一个 nil 对象
```

**nil 不是空指针，是一个真正分配出来的节点对象**，颜色涂成黑色，它自己的 left/right/parent 都指向自己。所有叶子节点的 left/right 都指向它，根节点的 parent 也指向它。

所以 `x->parent == T->nil` 的意思是：**"x 是根节点"**——因为根节点没有父节点，父指针指向了 nil。

**好处：**
1. 所有节点的指针永远非空，不用到处判空，不会段错误
2. nil 天然是黑色节点，满足"每条路径黑节点数相同"的性质
3. 统一判断：`x->left == T->nil` 代替 `x->left == nullptr`

| 要判断什么 | 普通树 | 红黑树 |
|---|---|---|
| 是不是叶子 | `x->left == nullptr` | `x->left == T->nil` |
| 是不是根节点 | `x->parent == nullptr` | `x->parent == T->nil` |
| 搜索到没找到 | `return nullptr` | `return T->nil` |

### 左旋/右旋

旋转是红黑树**保持平衡**的手段：

```
左旋：         右旋：
    x               y
     \             /
      y    →      x
```

旋转不改变 BST 性质（左 < 根 < 右），只是调整树的高度。

### 插入 fixup

新节点涂红色插入，如果父节点也是红色就违反红黑树性质，需要：
- 变色（叔叔红 → 父子变黑、祖父变红）
- 或旋转 + 变色

### 搜索就是 BST 查找

```
从根开始：
  key < node->key → 往左走
  key > node->key → 往右走
  key == node->key → 找到了
```

因为树是平衡的，高度是 log(n)，所以查找 O(log n)。

## 深入讨论：删除为什么比插入复杂？

| | 插入 | 删除 |
|---|---|---|
| 新节点颜色 | 固定涂红色 | 被删节点可能是黑色 |
| 破坏什么 | 只可能破坏"父子不能都红" | 破坏"黑高平衡" |
| 修复方向 | 往上走（从新节点到根） | 往上走，还要处理兄弟/侄子 |
| case 数量 | 2 种（叔叔红/黑） | 4 种（兄弟红黑 × 侄子红黑） |

### 删除四步流程

```
1. 找 y（真正要删的节点）
   - z 有 0/1 个子节点 → y = z
   - z 有 2 个子节点 → y = z 的后继（右子树最左节点）

2. 找 x（y 的孩子，代替 y 的位置）

3. 摘链：x 挂到 y 的父节点上

4. 如果删的是后继节点，把 y 的 key/value 换到 z（z 的内容不变，删的是后继）

5. 如果 y 是黑色 → delete_fixup 修复平衡
```

### 为什么有两个子节点时要删后继？

因为如果直接删 z，z 的两个子树就挂空了。标准做法：用后继节点（比 z 大的最小节点）替换 z 的值，然后删后继节点——后继节点最多只有一个子节点，删起来简单。

### delete_fixup 的 4 种 case

修复的核心：**x 是黑色的（相当于"多了一层黑"），要把这层黑"推"上去，直到根。**

```
case 1: 兄弟是红色 → 旋转，把兄弟变黑
case 2: 兄弟和两个侄子都是黑色 → 兄弟染红，x 往上推
case 3: 兄弟的右侄子是黑色 → 先旋转兄弟，转成 case 4
case 4: 兄弟的左侄子是红色 → 旋转父节点，搞定
```

左右对称两套，一共 8 个分支。

## 深入讨论：三个引擎接口统一

```
array:  kvs_array_set / kvs_array_get / kvs_array_delete ...
hash:   kvs_hash_set  / kvs_hash_get  / kvs_hash_delete ...
rbtree: kvs_rbtree_set / kvs_rbtree_get / kvs_rbtree_delete ...
```

**加新引擎 = 写一个 .cpp + 在命令表加几行。协议层 parser 一行都不用改。**

这就是第 7 步重构的价值——"对扩展开放，对修改关闭"。

## 常见错误：undefined reference to `g_tree`

如果你在头文件写了 `extern RbTree g_tree;` 但忘了在 .cpp 里定义，链接会报错：

```
undefined reference to `g_tree`
```

修复：在 `kvstore_rbtree.cpp` 里写一行：
```cpp
RbTree g_tree;   // 定义：真正分配内存
```

和 `extern` 声明配对使用。
